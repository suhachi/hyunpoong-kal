## 💡 주요 기술 구현

### 1. 동적 필터링
```typescript
function applyFilter() {
  if (filter === 'all') {
    setFilteredOrders(orders);
  } else if (filter === 'reviewable') {
    setFilteredOrders(getReviewableOrders(orders));
  } else {
    setFilteredOrders(filterOrdersByStatus(orders, filter));
  }
}

// filter 또는 orders가 변경될 때마다 자동 필터링
useEffect(() => {
  applyFilter();
}, [filter, orders]);
```

### 2. 타임스탬프 처리
```typescript
function formatDate(timestamp: any): string {
  let date: Date;
  
  // 여러 형식 지원
  if (typeof timestamp === 'string') {
    date = new Date(timestamp);
  } else if (timestamp?.seconds) {
    date = new Date(timestamp.seconds * 1000);
  } else if (timestamp?.toDate) {
    date = timestamp.toDate();
  } else {
    return '-';
  }

  // 상대적 시간 표시
  const diff = now.getTime() - date.getTime();
  const days = Math.floor(diff / (1000 * 60 * 60 * 24));

  if (days === 0) return '오늘 ' + time;
  if (days === 1) return '어제 ' + time;
  if (days < 7) return `${days}일 전`;
  return date.toLocaleDateString('ko-KR');
}
```

### 옵션 1: 주문내역 목록 페이지 생성 (권장)

**새로운 페이지 생성**: `/pages/app/OrderHistory.tsx`

```tsx
/**
 * 주문내역 목록 페이지
 * 고객의 모든 주문을 시간순으로 표시
 */

import { useState, useEffect } from 'react';
import { Link } from 'react-router-dom';
import { Card } from '../../components/ui/card';
import { Badge } from '../../components/ui/badge';
import { Button } from '../../components/ui/button';
import { getOrdersByUser } from '../../lib/orders.api';
import type { Order } from '../../types/order';

export default function OrderHistory() {
  const [orders, setOrders] = useState<Order[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    loadOrders();
  }, []);

  const loadOrders = async () => {
    try {
      // Mock 사용자 ID (실제로는 Auth에서 가져옴)
      const userId = 'mock-user-id';
      const data = await getOrdersByUser(userId);
      setOrders(data);
    } catch (error) {
      console.error('주문 목록 로딩 실패:', error);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="p-4 space-y-4 pb-24">
      <h1 className="text-2xl text-[#2E1C10] mb-4">주문내역</h1>
      
      {loading ? (
        <div>로딩 중...</div>
      ) : orders.length === 0 ? (
        <Card className="p-8 text-center">
          <p className="text-[#8B7355]">주문 내역이 없습니다</p>
        </Card>
      ) : (
        <div className="space-y-3">
          {orders.map((order) => (
            <Link 
              key={order.orderId} 
              to={`/order/${order.orderId}`}
              className="block"
            >
              <Card className="p-4 hover:shadow-md transition-shadow">
                <div className="flex justify-between items-start mb-2">
                  <div>
                    <p className="text-sm text-[#2E1C10] font-medium">
                      주문번호: {order.orderId}
                    </p>
                    <p className="text-xs text-[#8B7355]">
                      {new Date(order.createdAt.seconds * 1000).toLocaleDateString()}
                    </p>
                  </div>
                  <Badge>{order.status}</Badge>
                </div>
                
                <div className="text-sm text-[#2E1C10]">
                  {order.items[0].menuName} 
                  {order.items.length > 1 && ` 외 ${order.items.length - 1}개`}
                </div>
                
                <div className="flex justify-between items-center mt-2">
                  <p className="text-[#D61C1C] font-medium">
                    {order.finalAmount.toLocaleString()}원
                  </p>
                  {order.status === 'completed' && !order.reviewWritten && (
                    <Button size="sm" variant="outline">
                      리뷰 작성
                    </Button>
                  )}
                </div>
              </Card>
            </Link>
          ))}
        </div>
      )}
    </div>
  );
}
```

## 🔧 필요한 추가 작업

### 1. API 함수 추가

**파일**: `/lib/orders.api.ts` (새로 생성 필요)

```typescript
import { db, USE_FIREBASE } from './firebase';
import { collection, query, where, orderBy, getDocs } from 'firebase/firestore';
import type { Order } from '../types/order';

/**
 * 사용자의 주문 목록 조회
 */
export async function getOrdersByUser(userId: string): Promise<Order[]> {
  if (!USE_FIREBASE) {
    // Mock 데이터
    return [
      {
        orderId: 'ORD-001',
        userId: userId,
        items: [
          { menuId: 'M001', menuName: '현풍닭칼국수', quantity: 1, price: 9000 }
        ],
        totalAmount: 9000,
        deliveryFee: 3000,
        finalAmount: 12000,
        status: 'completed',
        createdAt: { seconds: Date.now() / 1000 - 86400 },
        phone: '010-1234-5678',
        deliveryType: 'delivery',
        payment: {
          method: 'card',
          status: 'paid'
        }
      }
    ];
  }

  try {
    const q = query(
      collection(db, 'orders'),
      where('userId', '==', userId),
      orderBy('createdAt', 'desc')
    );
    
    const snapshot = await getDocs(q);
    return snapshot.docs.map(doc => ({
      orderId: doc.id,
      ...doc.data()
    })) as Order[];
  } catch (error) {
    console.error('주문 목록 조회 실패:', error);
    return [];
  }
}
```

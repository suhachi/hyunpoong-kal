```javascript
window.__QA__=(()=>{
  const errors=[];
  const unhandled=[];
  const push=(arr,type,args)=>arr.push({
    type,
    args:[...args],
    time:new Date().toISOString(),
    path:location.pathname
  });
  const orig=console.error;
  console.error=function(...a){
    push(errors,'console.error',a);
    return orig.apply(console,a);
  };
  window.addEventListener('error',e=>push(errors,'error',[e.message,e.error]));
  window.addEventListener('unhandledrejection',e=>unhandled.push({
    reason:e.reason,
    time:new Date().toISOString(),
    path:location.pathname
  }));
  return{dump:()=>({errors,unhandled})};
})()
```

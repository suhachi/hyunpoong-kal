const handleToggleAvailability = async (menuId: string) => {
  if (!user) return;

  setActionLoading(true);
  try {
    // API 호출
    const updated = await toggleMenuAvailability(
      menuId, 
      user.uid, 
      user.name
    );
    
    // ✅ UI 즉시 반영

      prev.map(m => m.menuId === menuId ? updated : m)

## 🎯 6. 사용 시나리오

### 시나리오 1: 닭칼국수 품절 처리

**상황:**  
오후 2시, 닭 재고가 소진되어 닭칼국수를 품절 처리해야 함.

**절차:**
1. 관리자 → 메뉴 관리 페이지 접속
2. "닭칼국수" 행의 판매 스위치 OFF
3. ✅ **즉시 효과:**
   - 배지: "판매중" → "품절"
   - 색상: 녹색 → 회색
   - 스위치: ON → OFF

// firestore.rules
match /menus/{menuId} {
  // 읽기: 누구나
  allow read: if true;
  
  // 쓰기: 관리자만
  allow write: if request.auth != null 
    && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'owner';
  
  // 품절 토글: 관리자만 (특정 필드)
  allow update: if request.auth != null
    && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'owner'
    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['isAvailable', 'updatedAt', 'updatedBy']);
}

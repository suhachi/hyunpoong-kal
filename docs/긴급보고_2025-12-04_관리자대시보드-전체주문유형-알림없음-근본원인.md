### 방안 C: useOrderNotifications Hook 통합 (Best Practice)

**수정 파일**: `src/pages/admin/Dashboard.tsx`

```tsx
import { useOrderNotifications } from '@/hooks/useOrderNotifications';
import { collection, query, orderBy } from 'firebase/firestore';
import { db } from '@/lib/firebase';

export function Dashboard() {
  const [newOrders, setNewOrders] = useState<Order[]>([]);
  
  const ordersQuery = useMemo(
    () => query(collection(db, 'orders'), orderBy('createdAt', 'desc')),
    []
  );

  // ✅ Hook 사용으로 간결한 구현
  useOrderNotifications(ordersQuery, {
    role: 'admin',
    enabled: true,
    notifyAdded: true,
    notifyModified: false,
    play: (type, { id, order }) => {
      // ✅ 오디오 재생
      const audio = new Audio('/alert3.mp3');
      audio.play().catch(() => {
        // Fallback to beep
        const ctx = new AudioContext();
        const osc = ctx.createOscillator();
        osc.frequency.value = 800;
        osc.connect(ctx.destination);
        osc.start();
        osc.stop(ctx.currentTime + 0.5);
      });
    },
    toast: (msg, type, { id, order }) => {
      // ✅ 토스트 알림
      toast(
        <div>
          <div className="font-bold">🔔 {msg}</div>
          <div className="text-sm">{order.orderId} - {formatPrice(order.finalAmount)}</div>
          <div className="flex gap-2 mt-2">
            <Button size="sm" onClick={() => handleAccept(order)}>접수</Button>
            <Button size="sm" variant="outline">알림 끄기</Button>
          </div>
        </div>,
        { duration: Infinity }
      );
    }
  });

  return (
    <div>
      {/* 기존 UI */}
    </div>
  );
}
```

**장점**:
- ✅ Hook 재사용으로 코드 간결
- ✅ AdminOrderAlert와 동일한 알림 시스템
- ✅ 확장성/유지보수성 우수

**단점**:

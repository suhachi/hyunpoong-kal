```yaml
# .github/workflows/encoding-check.yml
name: Encoding Check

on: [push, pull_request]

jobs:
  check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Check line endings
        run: |
          if git ls-files --eol | grep 'w/crlf'; then
            echo "CRLF line endings found!"
            exit 1
          fi
      - name: Check encoding
        run: |
```

## 📎 부록: 사용 가능한 스크립트

### ✅ sonner 정규화 (안전, 사용 가능)
```powershell
# sonner-fix.ps1
Write-Host "=== Sonner Import Normalization ===" -ForegroundColor Cyan

$files = Get-ChildItem -Path src -Recurse -Include *.ts,*.tsx
$fixedCount = 0

foreach ($file in $files) {
    $content = Get-Content $file.FullName -Raw -Encoding UTF8
    if ($content -match "from ['\`"]sonner@2\.0\.3['\`"]") {
        $newContent = $content -replace "from ['\`"]sonner@2\.0\.3['\`"]", "from 'sonner'"
        Set-Content $file.FullName -Value $newContent -NoNewline -Encoding UTF8
        Write-Host "✓ Fixed: $($file.Name)" -ForegroundColor Green
        $fixedCount++
    }
}

Write-Host "`n=== Summary ===" -ForegroundColor Cyan
Write-Host "Fixed files: $fixedCount" -ForegroundColor Yellow
Write-Host "Done!" -ForegroundColor Green
```

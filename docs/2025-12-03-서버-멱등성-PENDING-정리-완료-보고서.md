#### 2-2. 주요 로직

```typescript
// 트랜잭션을 사용한 멱등성 보장
const result = await db.runTransaction(async (transaction) => {
  const orderDoc = await transaction.get(orderRef);
  const orderData = orderDoc.data();
  const paymentStatus = orderData.payment?.status;
  
  // 1. 멱등성 체크: 이미 승인된 경우 기존 결과 반환
  if (paymentStatus === "authorized" || paymentStatus === "approved") {
    console.log("[approvePayment] Already approved (idempotent)");
    return existingResult; // NICEPAY 재호출 없이 반환
  }

  // 2. 취소/실패된 주문 체크
  if (paymentStatus === "cancelled" || paymentStatus === "failed") {
    throw new HttpsError("failed-precondition", "이미 취소/실패된 주문입니다");
  }

  // 3. APP_CARD 체크
  if (paymentMethod !== "app_card") {
    throw new HttpsError("invalid-argument", "앱 결제 방식이 아닙니다");
  }

  // 4. NICEPAY 승인 & 상태 업데이트 (트랜잭션 내)
  const mockResult = { /* ... */ };
  transaction.update(orderRef, { /* ... */ });
  
  return mockResult;
});
```

### STEP 3. PENDING 주문 자동 취소 Scheduler 구현

#### 3-1. 정책
*   **대상**: `status === "pending"` AND `payment.method === "app_card"` AND `createdAt <= 30분 전`
*   **제외**: `MEET_CARD`, `MEET_CASH` (매장에서 직접 처리)
*   **실행**: 10분마다 (Asia/Seoul)

#### 3-2. 구현 파일
*   **`src/functions/src/schedulers/cleanup-pending-orders.ts`** (신규 생성)

#### 3-3. 주요 로직

```typescript
export const cleanupPendingOrders = functions
  .region(REGION)
  .pubsub.schedule("every 10 minutes")
  .timeZone("Asia/Seoul")
  .onRun(async (context) => {
    // 1) 30분 전 기준 시각 계산
    const cutoffTime = new Date(now.getTime() - 30 * 60 * 1000);
    
    // 2) 조건에 맞는 주문 조회
    const querySnapshot = await db
      .collection("orders")
      .where("status", "==", "pending")
      .where("payment.method", "==", "app_card")
      .where("createdAt", "<=", cutoffTimestamp)
      .get();
    
    // 3) 배치 업데이트로 취소 처리
    const batch = db.batch();
    for (const doc of querySnapshot.docs) {
      batch.update(doc.ref, {
        status: "cancelled",
        "payment.status": "failed",
        "payment.cancelReason": "payment_timeout",
        "meta.autoCanceledBy": "system",
        // ...
      });
    }
    await batch.commit();
  });
```

*   **이전**: 중복 승인 시 NICEPAY API를 여러 번 호출할 위험

### 2. PENDING 주문 자동 정리

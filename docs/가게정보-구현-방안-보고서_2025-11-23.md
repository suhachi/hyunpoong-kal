### 1단계: 가게 정보 API 생성

**파일**: `src/lib/admin/store.api.ts` (신규 생성)

**기능**:
- `getStoreInfo(storeId: string)`: 가게 정보 조회
- `updateStoreInfo(storeId: string, data: Partial<StoreDoc>)`: 가게 정보 업데이트
- Mock/Firebase 전환 지원

**구조**:
```typescript
import { USE_FIREBASE } from '../config/env';
import { db } from '../firebase';
import { storeDocRef, type StoreDoc } from '../firebase/firestore-schema';
import { getDoc, setDoc, serverTimestamp } from 'firebase/firestore';

// Mock 데이터
let mockStoreInfo: StoreDoc | null = null;

// Firebase: 가게 정보 조회
async function getStoreInfoFirebase(storeId: string): Promise<StoreDoc | null> {
  const docRef = storeDocRef(storeId);
  const docSnap = await getDoc(docRef);
  if (!docSnap.exists()) return null;
  return { storeId, ...docSnap.data() } as StoreDoc;
}

// Firebase: 가게 정보 업데이트
async function updateStoreInfoFirebase(
  storeId: string,
  data: Partial<StoreDoc>,
  by: string,
  byName: string
): Promise<StoreDoc> {
  const docRef = storeDocRef(storeId);
  const updateData = {
    ...data,
    updatedAt: serverTimestamp(),
    updatedBy: by,
    updatedByName: byName,
  };
  await setDoc(docRef, updateData, { merge: true });
  const updated = await getDoc(docRef);
  return { storeId, ...updated.data() } as StoreDoc;
}

// Public API
export async function getStoreInfo(storeId: string): Promise<StoreDoc | null> {
  if (USE_FIREBASE) {
    return await getStoreInfoFirebase(storeId);
  }
  // Mock 구현
  return mockStoreInfo;
}

export async function updateStoreInfo(
  storeId: string,
  data: Partial<StoreDoc>,
  by: string,
  byName: string
): Promise<StoreDoc> {
  if (USE_FIREBASE) {
    return await updateStoreInfoFirebase(storeId, data, by, byName);
  }
  // Mock 구현
  mockStoreInfo = { ...mockStoreInfo, ...data } as StoreDoc;
  return mockStoreInfo;
}
```

### 2단계: 관리자 설정에 가게 정보 탭 추가

**파일**: `src/pages/admin/Settings/StoreInfoTab.tsx` (신규 생성)

**기능**:
- 가게 이름 입력
- 주소 입력 (상세 주소, 위도/경도)
- 전화번호 입력
- 영업시간 설정
- 영업 상태 토글
- 배달 가능 여부 토글
- 최소 주문 금액 설정
- 배달비 설정

**UI 구조**:
```typescript
import { useState, useEffect } from 'react';
import { Card, CardContent, CardHeader, CardTitle } from '../../../components/ui/card';
import { Input } from '../../../components/ui/input';
import { Label } from '../../../components/ui/label';
import { Button } from '../../../components/ui/button';
import { Switch } from '../../../components/ui/switch';
import { getStoreInfo, updateStoreInfo } from '../../../lib/admin/store.api';
import { useAuth } from '../../../contexts/AuthContext';
import { STORE_ID } from '../../../config/env';
import { toast } from 'sonner';
import type { StoreDoc } from '../../../lib/firebase/firestore-schema';

export function StoreInfoTab() {
  const { user } = useAuth();
  const [storeInfo, setStoreInfo] = useState<StoreDoc | null>(null);
  const [loading, setLoading] = useState(false);

  // 가게 정보 로드
  useEffect(() => {
    loadStoreInfo();
  }, []);

  const loadStoreInfo = async () => {
    try {
      const info = await getStoreInfo(STORE_ID);
      if (info) {
        setStoreInfo(info);
      } else {
        // 기본값 설정
        setStoreInfo({
          storeId: STORE_ID,
          name: '',
          address: { full: '', detail: '' },
          phone: '',
          businessHours: {
            open: '10:00',
            close: '22:00',
            days: ['mon', 'tue', 'wed', 'thu', 'fri', 'sat', 'sun'],
          },
          isOpen: true,
          deliveryAvailable: true,
          minOrderAmount: 15000,
          deliveryFee: 3000,
          settings: {
            pointsRate: 0.03,
            pointsMinUse: 1000,
          },
          createdAt: new Date() as any,
          updatedAt: new Date() as any,
        });
      }
    } catch (error) {
      toast.error('가게 정보를 불러오는데 실패했습니다');
    }
  };

  const handleSave = async () => {
    if (!storeInfo || !user) return;
    setLoading(true);
    try {
      await updateStoreInfo(
        STORE_ID,
        storeInfo,
        user.uid,
        user.displayName || '관리자'
      );
      toast.success('가게 정보가 저장되었습니다');
    } catch (error) {
      toast.error('가게 정보 저장에 실패했습니다');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="space-y-6">
      <Card>
        <CardHeader>
          <CardTitle>기본 정보</CardTitle>
        </CardHeader>
        <CardContent className="space-y-4">
          <div>
            <Label>가게 이름 *</Label>
            <Input
              value={storeInfo?.name || ''}
              onChange={(e) => setStoreInfo({
                ...storeInfo!,
                name: e.target.value,
              })}
              placeholder="현풍닭칼국수"
            />
          </div>
          
          <div>
            <Label>전화번호 *</Label>
            <Input
              value={storeInfo?.phone || ''}
              onChange={(e) => setStoreInfo({
                ...storeInfo!,
                phone: e.target.value,
              })}
              placeholder="053-123-4567"
            />
          </div>
        </CardContent>
      </Card>

      <Card>
        <CardHeader>
          <CardTitle>주소 정보</CardTitle>
        </CardHeader>
        <CardContent className="space-y-4">
          <div>
            <Label>주소 *</Label>
            <Input
              value={storeInfo?.address?.full || ''}
              onChange={(e) => setStoreInfo({
                ...storeInfo!,
                address: {
                  ...storeInfo?.address,
                  full: e.target.value,
                },
              })}
              placeholder="대구광역시 달성군 현풍면"
            />
          </div>
          
          <div>
            <Label>상세 주소</Label>
            <Input
              value={storeInfo?.address?.detail || ''}
              onChange={(e) => setStoreInfo({
                ...storeInfo!,
                address: {
                  ...storeInfo?.address,
                  detail: e.target.value,
                },
              })}
              placeholder="상세 주소를 입력하세요"
            />
          </div>
        </CardContent>
      </Card>

      <Card>
        <CardHeader>
          <CardTitle>영업 설정</CardTitle>
        </CardHeader>
        <CardContent className="space-y-4">
          <div className="flex items-center justify-between">
            <Label>영업 중</Label>
            <Switch
              checked={storeInfo?.isOpen ?? true}
              onCheckedChange={(checked) => setStoreInfo({
                ...storeInfo!,
                isOpen: checked,
              })}
            />
          </div>
          
          <div className="flex items-center justify-between">
            <Label>배달 가능</Label>
            <Switch
              checked={storeInfo?.deliveryAvailable ?? true}
              onCheckedChange={(checked) => setStoreInfo({
                ...storeInfo!,
                deliveryAvailable: checked,
              })}
            />
          </div>
          
          <div>
            <Label>최소 주문 금액 (원)</Label>
            <Input
              type="number"
              value={storeInfo?.minOrderAmount || 15000}
              onChange={(e) => setStoreInfo({
                ...storeInfo!,
                minOrderAmount: parseInt(e.target.value) || 0,
              })}
            />
          </div>
          
          <div>
            <Label>기본 배달비 (원)</Label>
            <Input
              type="number"
              value={storeInfo?.deliveryFee || 3000}
              onChange={(e) => setStoreInfo({
                ...storeInfo!,
                deliveryFee: parseInt(e.target.value) || 0,
              })}
            />
          </div>
        </CardContent>
      </Card>

      <div className="flex justify-end">
        <Button onClick={handleSave} disabled={loading}>
          저장
        </Button>
      </div>
    </div>
  );
}
```

import { getStoreInfo } from '../../lib/admin/store.api';
import { STORE_ID } from '../../config/env';
import type { StoreDoc } from '../../lib/firebase/firestore-schema';

export function Home() {
  const [storeInfo, setStoreInfo] = useState<StoreDoc | null>(null);

  useEffect(() => {
    const loadStoreInfo = async () => {
      try {
        const info = await getStoreInfo(STORE_ID);
        setStoreInfo(info);
      } catch (error) {
        console.error('Failed to load store info:', error);
      }
    };
    loadStoreInfo();
  }, []);

  return (
    <div className="space-y-6">
      <section className="relative h-[300px] bg-gradient-to-b from-[#D61C1C] to-[#F37021]/20 mt-6">
        <div className="absolute inset-0 flex flex-col items-center justify-center text-white p-8">
          <h1 className="text-3xl mb-4 text-center drop-shadow-lg">
            {storeInfo?.name || '현풍닭칼국수'}
          </h1>
          <p className="text-lg text-center drop-shadow-md opacity-90">
            {storeInfo?.address?.full || '정성껏 끓여낸 진한 국물과 쫄깃한 수타면'}
          </p>
          {/* ... */}
        </div>
      </section>
      
      {/* 영업 상태 */}
      <div className="flex items-center gap-2 p-4 bg-white rounded-2xl shadow-sm">
        <div className="flex items-center justify-center w-3 h-3">
          <span className={`w-full h-full rounded-full animate-pulse ${
            storeInfo?.isOpen ? 'bg-green-500' : 'bg-gray-400'
          }`} />
        </div>
        <span className="text-sm text-[#2E1C10]">
          {storeInfo?.isOpen ? '영업중' : '영업 종료'}
        </span>
        {storeInfo?.businessHours && (
          <span className="text-sm text-[#2E1C10]/60">
            {storeInfo.businessHours.open} - {storeInfo.businessHours.close}
          </span>
        )}
      </div>
      {/* ... */}
    </div>
  );
}
